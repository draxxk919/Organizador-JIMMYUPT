function iniciarSincronizacion( usuario ){ if( detenerSincronizacion ){ detenerSincronizacion(); detenerSincronizacion = null; } const referencia = db .collection("usuarios") .doc(usuario.uid) .collection("datos") .doc("principal"); detenerSincronizacion = referencia.onSnapshot( function(documento){ if( documento.exists ){ const nube = documento.data(); cargandoNube = true; datos.cursos = Array.isArray( nube.cursos ) ? nube.cursos : []; datos.tareas = Array.isArray( nube.tareas ) ? nube.tareas : []; datos.notas = Array.isArray( nube.notas ) ? nube.notas : []; if( nube.tema ){ datos.tema = nube.tema; } try { localStorage.setItem( STORAGE, JSON.stringify( datos ) ); } catch(error){ console.error( "No se pudo actualizar la copia local:", error ); } actualizar(); cargandoNube = false; } else{ referencia.set({ cursos: datos.cursos || [], tareas: datos.tareas || [], notas: datos.notas || [], tema: datos.tema || "claro" }) .catch( function(error){ console.error( "No se pudieron crear los datos iniciales en Firebase:", error ); } ); } }, function(error){ console.error( "Error de sincronización:", error ); } ); }
